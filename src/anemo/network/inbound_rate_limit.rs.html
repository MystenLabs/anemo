1use std::collections::HashMap;
2use std::net::IpAddr;
3use std::num::NonZeroU32;
4use std::time::Instant;
5
6/// Hard cap on the number of distinct source IPs tracked at once.
7/// At a few dozen bytes per entry this bounds the map near ~2 MiB.
8const MAX_TRACKED_IPS: usize = 16_384;
9
10/// A token-bucket rate limiter keyed by source IP, used to bound the rate at which
11/// new inbound connections from any single source are admitted to the TLS handshake.
12pub(crate) struct InboundIpRateLimiter {
13    /// Sustained refill rate, in tokens (connections) per second.
14    rate_per_sec: f64,
15    /// Bucket capacity, i.e. the largest instantaneous burst.
16    burst: f64,
17    buckets: HashMap<IpAddr, TokenBucket>,
18}
19
20struct TokenBucket {
21    tokens: f64,
22    last_refill: Instant,
23}
24
25impl InboundIpRateLimiter {
26    pub(crate) fn new(rate_per_sec: NonZeroU32, burst: NonZeroU32) -> Self {
27        Self {
28            rate_per_sec: f64::from(rate_per_sec.get()),
29            burst: f64::from(burst.get()),
30            buckets: HashMap::new(),
31        }
32    }
33
34    /// Returns `true` if a new inbound connection from `ip` is permitted, consuming
35    /// one token. `now` is taken as an argument to keep the type unit-testable.
36    pub(crate) fn check(&mut self, ip: IpAddr, now: Instant) -> bool {
37        // Enforce size cap. Idle buckets are reclaimed separately by `evict_idle`.
38        if self.buckets.len() >= MAX_TRACKED_IPS && !self.buckets.contains_key(&ip) {
39            return false;
40        }
41
42        let rate = self.rate_per_sec;
43        let burst = self.burst;
44        let bucket = self.buckets.entry(ip).or_insert(TokenBucket {
45            tokens: burst,
46            last_refill: now,
47        });
48
49        bucket.tokens = Self::refilled_tokens(bucket, now, rate, burst);
50        bucket.last_refill = now;
51
52        if bucket.tokens >= 1.0 {
53            bucket.tokens -= 1.0;
54            true
55        } else {
56            false
57        }
58    }
59
60    /// Drop buckets that have refilled to capacity (i.e. have seen no recent
61    /// activity); a full bucket is indistinguishable from a never-seen IP, so
62    /// forgetting it is safe.
63    pub(crate) fn evict_idle(&mut self, now: Instant) {
64        let rate = self.rate_per_sec;
65        let burst = self.burst;
66        self.buckets
67            .retain(|_ip, bucket| Self::refilled_tokens(bucket, now, rate, burst) < burst);
68    }
69
70    /// Tokens `bucket` holds after refilling for the time elapsed since its last
71    /// update, clamped to the burst capacity.
72    fn refilled_tokens(bucket: &TokenBucket, now: Instant, rate: f64, burst: f64) -> f64 {
73        let elapsed = now
74            .saturating_duration_since(bucket.last_refill)
75            .as_secs_f64();
76        (bucket.tokens + elapsed * rate).min(burst)
77    }
78
79    #[cfg(test)]
80    fn tracked_ips(&self) -> usize {
81        self.buckets.len()
82    }
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88    use std::net::Ipv4Addr;
89    use std::time::Duration;
90
91    fn ip(n: u8) -> IpAddr {
92        IpAddr::V4(Ipv4Addr::new(127, 0, 0, n))
93    }
94
95    fn nz(n: u32) -> NonZeroU32 {
96        NonZeroU32::new(n).unwrap()
97    }
98
99    #[test]
100    fn admits_full_burst_then_throttles() {
101        let now = Instant::now();
102        let mut limiter = InboundIpRateLimiter::new(nz(10), nz(10));
103        for _ in 0..10 {
104            assert!(limiter.check(ip(1), now));
105        }
106        // The next attempt at the same instant is over budget.
107        assert!(!limiter.check(ip(1), now));
108    }
109
110    #[test]
111    fn refills_at_the_configured_rate() {
112        let now = Instant::now();
113        let mut limiter = InboundIpRateLimiter::new(nz(10), nz(10));
114        for _ in 0..10 {
115            assert!(limiter.check(ip(1), now));
116        }
117        assert!(!limiter.check(ip(1), now));
118
119        // After half a second, ~5 tokens (0.5s * 10/s) have refilled.
120        let later = now + Duration::from_millis(500);
121        let admitted = (0..10).filter(|_| limiter.check(ip(1), later)).count();
122        assert_eq!(admitted, 5);
123    }
124
125    #[test]
126    fn buckets_are_per_ip() {
127        let now = Instant::now();
128        let mut limiter = InboundIpRateLimiter::new(nz(1), nz(1));
129        assert!(limiter.check(ip(1), now));
130        assert!(!limiter.check(ip(1), now));
131        // A different source IP has an independent bucket.
132        assert!(limiter.check(ip(2), now));
133        assert!(!limiter.check(ip(2), now));
134    }
135
136    #[test]
137    fn evict_idle_reclaims_fully_refilled_buckets() {
138        let now = Instant::now();
139        let mut limiter = InboundIpRateLimiter::new(nz(10), nz(10));
140        for n in 0u8..50 {
141            assert!(limiter.check(ip(n), now));
142        }
143        assert_eq!(limiter.tracked_ips(), 50);
144
145        // After enough time for the buckets to refill to capacity, a sweep reclaims
146        // them all -- they're indistinguishable from never-seen IPs.
147        limiter.evict_idle(now + Duration::from_secs(60));
148        assert_eq!(limiter.tracked_ips(), 0);
149    }
150
151    #[test]
152    fn tracked_ips_never_exceeds_the_cap() {
153        let now = Instant::now();
154        // rate = burst = 1 leaves every bucket drained (non-full) at `now`, so eviction
155        // can reclaim nothing and the cap must be held by rejecting new sources.
156        let mut limiter = InboundIpRateLimiter::new(nz(1), nz(1));
157        for n in 0..MAX_TRACKED_IPS as u32 {
158            assert!(limiter.check(IpAddr::V4(Ipv4Addr::from(n)), now));
159        }
160        assert_eq!(limiter.tracked_ips(), MAX_TRACKED_IPS);
161
162        // A further distinct source is rejected rather than growing the map past the cap.
163        let overflow = IpAddr::V4(Ipv4Addr::from(MAX_TRACKED_IPS as u32));
164        assert!(!limiter.check(overflow, now));
165        assert_eq!(limiter.tracked_ips(), MAX_TRACKED_IPS);
166    }
167
168    #[test]
169    fn idle_refill_is_capped_at_burst() {
170        let now = Instant::now();
171        let mut limiter = InboundIpRateLimiter::new(nz(2), nz(3));
172        for _ in 0..3 {
173            assert!(limiter.check(ip(1), now));
174        }
175        assert!(!limiter.check(ip(1), now));
176
177        // A long idle period refills only up to the burst capacity, not beyond.
178        let later = now + Duration::from_secs(100);
179        let admitted = (0..10).filter(|_| limiter.check(ip(1), later)).count();
180        assert_eq!(admitted, 3);
181    }

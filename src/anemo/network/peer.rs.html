15/// Handle to a connection with a remote Peer.
16#[derive(Clone)]
17pub struct Peer {

48    pub async fn rpc(&mut self, request: Request<Bytes>) -> Result<Response<Bytes>> {
49        self.ready().await?.call(request).await
50    }
51
52    async fn do_rpc(&self, request: Request<Bytes>) -> Result<Response<Bytes>> {
53        let (send_stream, recv_stream) = self.connection.open_bi().await?;
54        let mut send_stream =
55            FramedWrite::new(send_stream, request_message_frame_codec(&self.config));
56        let mut recv_stream =
57            FramedRead::new(recv_stream, response_message_frame_codec(&self.config));
58
59        //
60        // Write Request
61        //
62
63        write_request(&mut send_stream, request).await?;
64        send_stream.get_mut().finish()?;
65
66        //
67        // Read Response
68        //
69
70        let mut response = read_response(&mut recv_stream).await?;
71
72        // Set the PeerId of this peer
73        response.extensions_mut().insert(self.peer_id());
74
75        Ok(response)
76    }
77}

79impl Service<Request<Bytes>> for Peer {
80    type Response = Response<Bytes>;
81    type Error = crate::Error;
82    type Future = BoxFuture<'static, Result<Self::Response, Self::Error>>;
83
84    #[inline]
85    fn poll_ready(
86        &mut self,
87        _: &mut std::task::Context<'_>,
88    ) -> std::task::Poll<Result<(), Self::Error>> {
89        std::task::Poll::Ready(Ok(()))
90    }
91
92    #[inline]
93    fn call(&mut self, mut request: Request<Bytes>) -> Self::Future {
94        // Provide Connection Metadata to middleware layers via extensions including:
95        // * PeerId
96        // * Direction of the Request
97        request.extensions_mut().insert(self.peer_id());
98        request.extensions_mut().insert(crate::Direction::Outbound);
99
100        let peer = self.clone();
101        let inner = tower::service_fn(move |request| {
102            let peer = peer.clone();
103            async move { peer.do_rpc(request).await }
104        })
105        .boxed();
106
107        let mut service = self.outbound_request_layer.layer(inner);
108        service.call(request)
109    }

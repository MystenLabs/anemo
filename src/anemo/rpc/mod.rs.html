13pub struct Status {
14    status: StatusCode,
15    headers: HeaderMap,
16    message: Option<String>,
17    peer_id: Option<PeerId>,
18
19    /// Optional underlying error.
20    source: Option<BoxError>,
21}
22
23impl Status {
24    /// Create a new `Status` with the associated code and message.
25    pub fn new(status: StatusCode) -> Self {
26        Self {
27            status,
28            message: None,
29            peer_id: None,
30            headers: HeaderMap::default(),
31            source: None,
32        }
33    }
34
35    pub fn new_with_message<M: Into<String>>(status: StatusCode, message: M) -> Self {
36        let mut status = Self::new(status);
37        status.message = Some(message.into());
38        status
39    }
40
41    pub fn unknown<M: Into<String>>(message: M) -> Self {
42        Self::new_with_message(StatusCode::Unknown, message)
43    }
44
45    pub fn internal<M: Into<String>>(message: M) -> Self {
46        Self::new_with_message(StatusCode::InternalServerError, message)
47    }
48
49    pub fn from_error(error: BoxError) -> Status {
50        let mut status = Self::new(StatusCode::Unknown);
51        status.message = Some(format!("unknown error: {error}"));
52        status.source = Some(error);
53        status
54    }
55
56    pub fn status(&self) -> StatusCode {
57        self.status
58    }
59
60    pub fn headers(&self) -> &HeaderMap {
61        &self.headers
62    }
63
64    pub fn headers_mut(&mut self) -> &mut HeaderMap {
65        &mut self.headers
66    }
67
68    pub fn with_header<K: Into<String>, V: Into<String>>(mut self, key: K, value: V) -> Self {
69        self.headers_mut().insert(key.into(), value.into());
70        self
71    }
72
73    pub fn peer_id(&self) -> Option<&PeerId> {
74        self.peer_id.as_ref()
75    }
76
77    fn from_response<T>(response: Response<T>) -> Self {
78        let peer_id = response.peer_id().copied();
79        let (parts, _body) = response.into_parts();
80
81        let message = parts
82            .headers
83            .get(crate::types::header::STATUS_MESSAGE)
84            .cloned();
85        Self {
86            status: parts.status,
87            message,
88            peer_id,
89            headers: parts.headers,
90            source: None,
91        }
92    }
93}

95impl IntoResponse for Status {
96    fn into_response(self) -> Response<bytes::Bytes> {
97        let mut response = self.status.into_response();
98
99        response.headers_mut().extend(self.headers);
100
101        if let Some(message) = self.message {
102            response
103                .headers_mut()
104                .insert(crate::types::header::STATUS_MESSAGE.to_owned(), message);
105        }
106
107        response
108    }
109}

121    pub struct Rpc<T> {
122        inner: T,
123    }
124
125    impl<T> Rpc<T> {
126        pub fn new(inner: T) -> Self {
127            Self { inner }
128        }
129
130        /// Gets a reference to the underlying service.
131        pub fn inner(&self) -> &T {
132            &self.inner
133        }
134
135        /// Gets a mutable reference to the underlying service.
136        pub fn inner_mut(&mut self) -> &mut T {
137            &mut self.inner
138        }
139
140        /// Consumes `self`, returning the underlying service.
141        pub fn into_inner(self) -> T {
142            self.inner
143        }
144
145        pub async fn ready(&mut self) -> Result<(), T::Error>
146        where
147            T: Service<Request<Bytes>>,
148        {
149            futures::future::poll_fn(|cx| self.inner.poll_ready(cx)).await
150        }
151
152        /// Send a single unary RPC request.
153        pub async fn unary<M1, M2, C>(
154            &mut self,
155            request: Request<M1>,
156            mut codec: C,
157        ) -> Result<Response<M2>, Status>
158        where
159            T: Service<Request<Bytes>, Response = Response<Bytes>>,
160            T::Error: Into<BoxError>,
161            C: Codec<Encode = M1, Decode = M2>,
162            M1: Send + Sync + 'static,
163            M2: Send + Sync + 'static,
164        {
165            let request = {
166                let (mut parts, body) = request.into_parts();
167
168                // Set the content type
169                parts.headers.insert(
170                    crate::types::header::CONTENT_TYPE.to_owned(),
171                    codec.format_name().to_owned(),
172                );
173
174                let mut encoder = codec.encoder();
175                let bytes = encoder
176                    .encode(body)
177                    .map_err(Into::into)
178                    .map_err(Status::from_error)?;
179
180                Request::from_parts(parts, bytes)
181            };
182
183            let response = self
184                .inner
185                .call(request)
186                .await
187                .map_err(Into::into)
188                .map_err(Status::from_error)?;
189
190            let status_code = response.status();
191
192            if !status_code.is_success() {
193                return Err(Status::from_response(response));
194            }
195
196            let response = {
197                let (parts, body) = response.into_parts();
198
199                let mut decoder = codec.decoder();
200                let message = decoder
201                    .decode(body)
202                    .map_err(Into::into)
203                    .map_err(Status::from_error)?;
204
205                Response::from_parts(parts, message)
206            };
207
208            Ok(response)
209        }
210    }

223    use std::future::Future;
224
225    pub struct Rpc<T1, T2> {
226        request_codec: T1,
227        response_codec: T2,
228    }
229
230    impl<T1, T2> Rpc<T1, T2>
231    where
232        T1: Codec,
233        T2: Codec,
234    {
235        pub fn new(request_codec: T1, response_codec: T2) -> Self {
236            Self {
237                request_codec,
238                response_codec,
239            }
240        }
241
242        /// Handle a single unary RPC request.
243        pub async fn unary<S>(&mut self, mut service: S, request: Request<Bytes>) -> Response<Bytes>
244        where
245            S: UnaryService<T1::Decode, Response = T2::Encode>,
246        {
247            let request = match self.map_request(request).await {
248                Ok(r) => r,
249                Err(status) => {
250                    return self.map_response(Err(status));
251                }
252            };
253
254            let response = service.call(request).await;
255
256            self.map_response(response)
257        }
258
259        async fn map_request(
260            &mut self,
261            request: Request<Bytes>,
262        ) -> Result<Request<T1::Decode>, Status> {
263            let (parts, body) = request.into_parts();
264
265            let mut decoder = self.request_codec.decoder();
266            let message = decoder
267                .decode(body)
268                .map_err(Into::into)
269                .map_err(Status::from_error)?;
270
271            let req = Request::from_parts(parts, message);
272
273            Ok(req)
274        }
275
276        fn map_response(
277            &mut self,
278            response: Result<crate::Response<T2::Encode>, Status>,
279        ) -> Response<Bytes> {
280            let response = match response {
281                Ok(r) => r,
282                Err(status) => return status.into_response(),
283            };
284
285            let (mut parts, body) = response.into_parts();
286
287            // Set the content type
288            parts.headers.insert(
289                crate::types::header::CONTENT_TYPE.to_owned(),
290                self.response_codec.format_name().to_owned(),
291            );
292
293            let mut encoder = self.response_codec.encoder();
294            let bytes = match encoder
295                .encode(body)
296                .map_err(Into::into)
297                .map_err(|err| Status::internal(format!("Error encoding: {err}")))
298            {
299                Ok(bytes) => bytes,
300                Err(status) => return status.into_response(),
301            };
302
303            Response::from_parts(parts, bytes)
304        }
305    }

307    /// A specialization of tower_service::Service.
308    ///
309    /// Existing tower_service::Service implementations with the correct form will
310    /// automatically implement `UnaryService`.
311    pub trait UnaryService<R> {
312        /// Response message type
313        type Response;
314
315        /// Response future
316        type Future: Future<Output = Result<Response<Self::Response>, Status>>;
317
318        /// Call the service
319        fn call(&mut self, request: Request<R>) -> Self::Future;
320    }
321
322    impl<T, M1, M2> UnaryService<M1> for T
323    where
324        T: Service<Request<M1>, Response = Response<M2>, Error = Status>,
325    {
326        type Response = M2;
327        type Future = T::Future;
328
329        fn call(&mut self, request: Request<M1>) -> Self::Future {
330            Service::call(self, request)
331        }
332    }
333}

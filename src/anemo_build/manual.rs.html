1//! This module provides utilities for generating `anemo` service stubs and clients.
2//!
3//! # Example
4//!
5//! ```rust,no_run
6//! fn main() -> Result<(), Box<dyn std::error::Error>> {
7//!     let greeter_service = anemo_build::manual::Service::builder()
8//!         .name("Greeter")
9//!         .package("helloworld")
10//!         .method(
11//!             anemo_build::manual::Method::builder()
12//!                 .name("say_hello")
13//!                 .route_name("SayHello")
14//!                 // Provide the path to the Request type
15//!                 .request_type("crate::HelloRequest")
16//!                 // Provide the path to the Response type
17//!                 .response_type("super::HelloResponse")
18//!                 // Provide the path to the Codec to use
19//!                 .codec_path("crate::JsonCodec")
20//!                 .build(),
21//!         )
22//!         .build();
23//!
24//!     anemo_build::manual::Builder::new().compile(&[greeter_service]);
25//!     Ok(())
26//! }
27//! ```

37/// Service builder.
38///
39/// This builder can be used to manually define an RPC service.
40///
41/// # Example
42///
43/// ```
44/// # use anemo_build::manual::Service;
45/// let greeter_service = Service::builder()
46///     .name("Greeter")
47///     .package("helloworld")
48///     // Add various methods to the service
49///     // .method()
50///     .build();
51/// ```
52#[derive(Debug, Default)]
53pub struct ServiceBuilder {
54    /// The service name in Rust style.
55    name: Option<String>,
56    /// The package name.
57    package: Option<String>,
58    /// The service comments.
59    comments: Vec<String>,
60    /// The service methods.
61    methods: Vec<Method>,
62    /// Attributes to apply to generated code.
63    attributes: Attributes,
64}
65
66impl ServiceBuilder {
67    /// Set the name for this Service.
68    ///
69    /// This value will be used both as the base for the generated rust types and service trait as
70    /// well as part of the route for calling this service. Routes have the form:
71    /// `/<package_name>.<service_name>/<method_route_name>`
72    pub fn name(mut self, name: impl AsRef<str>) -> Self {
73        self.name = Some(name.as_ref().to_owned());
74        self
75    }
76
77    /// Set the package this Service is part of.
78    ///
79    /// This value will be used as part of the route for calling this service.
80    /// Routes have the form: `/<package_name>.<service_name>/<method_route_name>`
81    pub fn package(mut self, package: impl AsRef<str>) -> Self {
82        self.package = Some(package.as_ref().to_owned());
83        self
84    }
85
86    /// Add a comment string that should be included as a doc comment for this Service.
87    pub fn comment(mut self, comment: impl AsRef<str>) -> Self {
88        self.comments.push(comment.as_ref().to_owned());
89        self
90    }
91
92    /// Adds a Method to this Service.
93    pub fn method(mut self, method: Method) -> Self {
94        self.methods.push(method);
95        self
96    }
97
98    /// Adds Attributes which should be applied to items in the generated code.
99    pub fn attributes(mut self, attributes: Attributes) -> Self {
100        self.attributes = attributes;
101        self
102    }
103
104    /// Build a Service.
105    ///
106    /// Panics if `name` wasn't set.
107    pub fn build(self) -> Service {
108        Service {
109            name: self.name.unwrap(),
110            comments: self.comments,
111            package: self.package.unwrap_or_default(),
112            methods: self.methods,
113            attributes: self.attributes,
114        }
115    }
116}

164/// A service method descriptor.
165#[derive(Debug)]
166pub struct Method {
167    /// The name of the method in Rust style.
168    name: String,
169    /// The name of the method as should be used when constructing a route
170    route_name: String,
171    /// The method comments.
172    comments: Vec<String>,
173    /// The input Rust type.
174    request_type: String,
175    /// The output Rust type.
176    response_type: String,
177    /// The path to the codec to use for this method
178    codec_path: String,
179    /// Use raw (serialized) bytes for the server-side response handler.
180    server_handler_return_raw_bytes: bool,
181}
182
183impl Method {
184    /// Create a new `MethodBuilder`
185    pub fn builder() -> MethodBuilder {
186        MethodBuilder::default()
187    }
188
189    pub fn name(&self) -> &str {
190        &self.name
191    }
192
193    pub fn identifier(&self) -> &str {
194        &self.route_name
195    }
196
197    pub fn codec_path(&self) -> &str {
198        &self.codec_path
199    }
200
201    pub fn comment(&self) -> &[String] {
202        &self.comments
203    }
204
205    pub fn request_type(&self) -> TokenStream {
206        syn::parse_str::<syn::Type>(&self.request_type)
207            .unwrap()
208            .to_token_stream()
209    }
210
211    pub fn response_type(&self) -> TokenStream {
212        syn::parse_str::<syn::Type>(&self.response_type)

216

218        self.server_handler_return_raw_bytes
219    }

222/// Method builder.
223///
224/// This builder can be used to manually define an RPC method, which can be added to an RPC service.
225///
226/// # Example
227///
228/// ```
229/// # use anemo_build::manual::Method;
230/// let say_hello_method = Method::builder()
231///     .name("say_hello")
232///     .route_name("SayHello")
233///     // Provide the path to the Request type
234///     .request_type("crate::common::HelloRequest")
235///     // Provide the path to the Response type
236///     .response_type("crate::common::HelloResponse")
237///     // Provide the path to the Codec to use
238///     .codec_path("crate::common::JsonCodec")
239///     .build();
240/// ```
241#[derive(Debug, Default)]
242pub struct MethodBuilder {
243    /// The name of the method in Rust style.
244    name: Option<String>,
245    /// The name of the method as should be used when constructing a route
246    route_name: Option<String>,
247    /// The method comments.
248    comments: Vec<String>,
249    /// The input Rust type.
250    request_type: Option<String>,
251    /// The output Rust type.
252    response_type: Option<String>,
253    /// The path to the codec to use for this method
254    codec_path: Option<String>,
255    /// Use raw (serialized) bytes for the server-side response handler.
256    server_handler_return_raw_bytes: bool,
257}
258
259impl MethodBuilder {
260    /// Set the name for this Method.
261    ///
262    /// This value will be used for generating the client functions for calling this Method.
263    ///
264    /// Generally this is formatted in snake_case.
265    pub fn name(mut self, name: impl AsRef<str>) -> Self {
266        self.name = Some(name.as_ref().to_owned());
267        self
268    }
269
270    /// Set the route_name for this Method.
271    ///
272    /// This value will be used as part of the route for calling this method.
273    /// Routes have the form: `/<package_name>.<service_name>/<method_route_name>`
274    ///
275    /// Generally this is formatted in PascalCase.
276    pub fn route_name(mut self, route_name: impl AsRef<str>) -> Self {
277        self.route_name = Some(route_name.as_ref().to_owned());
278        self
279    }
280
281    /// Add a comment string that should be included as a doc comment for this Method.
282    pub fn comment(mut self, comment: impl AsRef<str>) -> Self {
283        self.comments.push(comment.as_ref().to_owned());
284        self
285    }
286
287    /// Set the path to the Rust type that should be use for the Request type of this method.
288    pub fn request_type(mut self, request_type: impl AsRef<str>) -> Self {
289        self.request_type = Some(request_type.as_ref().to_owned());
290        self
291    }
292
293    /// Set the path to the Rust type that should be use for the Response type of this method.
294    pub fn response_type(mut self, response_type: impl AsRef<str>) -> Self {
295        self.response_type = Some(response_type.as_ref().to_owned());
296        self
297    }
298
299    /// Set the path to the Rust type that should be used as the `Codec` for this method.
300    ///
301    /// Currently the codegen assumes that this type implements `Default`.
302    pub fn codec_path(mut self, codec_path: impl AsRef<str>) -> Self {
303        self.codec_path = Some(codec_path.as_ref().to_owned());
304        self
305    }
306
307    /// Set whether or not the server handler should use raw bytes for the response.
308    pub fn server_handler_return_raw_bytes(mut self, use_raw_bytes: bool) -> Self {
309        self.server_handler_return_raw_bytes = use_raw_bytes;
310        self
311    }
312
313    /// Build a Method
314    ///
315    /// Panics if `name`, `route_name`, `request_type`, `response_type`, or `codec_path` weren't set.
316    pub fn build(self) -> Method {
317        Method {
318            name: self.name.unwrap(),
319            route_name: self.route_name.unwrap(),
320            comments: self.comments,
321            request_type: self.request_type.unwrap(),
322            response_type: self.response_type.unwrap(),
323            codec_path: self.codec_path.unwrap(),
324            server_handler_return_raw_bytes: self.server_handler_return_raw_bytes,
325        }
326    }
327}

329struct ServiceGenerator {
330    builder: Builder,
331    clients: TokenStream,
332    servers: TokenStream,
333}
334
335impl ServiceGenerator {
336    fn generate(&mut self, service: &Service) {
337        if self.builder.build_server {
338            let server = server::generate(service);
339            self.servers.extend(server);
340        }
341
342        if self.builder.build_client {
343            let client = client::generate(service);
344            self.clients.extend(client);
345        }
346    }
347
348    fn finalize(&mut self, buf: &mut String) {
349        if self.builder.build_client && !self.clients.is_empty() {
350            let clients = &self.clients;
351
352            let client_service = quote::quote! {
353                #clients
354            };
355
356            let ast: syn::File = syn::parse2(client_service).expect("not a valid tokenstream");
357            let code = prettyplease::unparse(&ast);
358            buf.push_str(&code);
359
360            self.clients = TokenStream::default();
361        }
362
363        if self.builder.build_server && !self.servers.is_empty() {
364            let servers = &self.servers;
365
366            let server_service = quote::quote! {
367                #servers
368            };
369
370            let ast: syn::File = syn::parse2(server_service).expect("not a valid tokenstream");
371            let code = prettyplease::unparse(&ast);
372            buf.push_str(&code);
373
374            self.servers = TokenStream::default();
375        }
376    }
377}

379/// Service generator builder.
380#[derive(Debug)]
381pub struct Builder {
382    build_server: bool,
383    build_client: bool,
384
385    out_dir: Option<PathBuf>,
386}
387
388impl Default for Builder {
389    fn default() -> Self {
390        Self {
391            build_server: true,
392            build_client: true,
393            out_dir: None,
394        }
395    }
396}
397
398impl Builder {
399    /// Create a new Builder
400    pub fn new() -> Self {
401        Self::default()
402    }
403
404    /// Enable or disable RPC client code generation.
405    ///
406    /// Defaults to enabling client code generation.
407    pub fn build_client(mut self, enable: bool) -> Self {
408        self.build_client = enable;
409        self
410    }
411
412    /// Enable or disable RPC server code generation.
413    ///
414    /// Defaults to enabling server code generation.
415    pub fn build_server(mut self, enable: bool) -> Self {
416        self.build_server = enable;
417        self
418    }
419
420    /// Set the output directory to generate code to.
421    ///
422    /// Defaults to the `OUT_DIR` environment variable.
423    pub fn out_dir(mut self, out_dir: impl AsRef<Path>) -> Self {
424        self.out_dir = Some(out_dir.as_ref().to_path_buf());
425        self
426    }
427
428    /// Performs code generation for the provided services.
429    ///
430    /// Generated services will be output into the directory specified by `out_dir`
431    /// with files named `<package_name>.<service_name>.rs`.
432    pub fn compile(self, services: &[Service]) {
433        let out_dir = if let Some(out_dir) = self.out_dir.as_ref() {
434            out_dir.clone()
435        } else {
436            PathBuf::from(std::env::var("OUT_DIR").unwrap())
437        };
438
439        let mut generator = ServiceGenerator {
440            builder: self,
441            clients: TokenStream::default(),
442            servers: TokenStream::default(),
443        };
444
445        for service in services {
446            generator.generate(service);
447            let mut output = String::new();
448            generator.finalize(&mut output);
449
450            let out_file = out_dir.join(format!(
451                "{}{}{}.rs",
452                service.package,
453                if service.package.is_empty() { "" } else { "." },
454                service.name
455            ));
456            fs::write(out_file, output).unwrap();
457        }
458    }

1use std::convert::Infallible;

3use anemo::{Network, Peer, Request, Response, Result};

7async fn noop_handle(_request: Request<Bytes>) -> Result<Response<Bytes>, Infallible> {
8    Ok(Response::new(bytes::Bytes::new()))
9}
10
11pub async fn create_client_network(address: &str, server_name: &str) -> Result<(Network, Peer)> {
12    let network = Network::bind("0.0.0.0:0")
13        .private_key(random_key())
14        .server_name(server_name)
15        .outbound_request_layer(TraceLayer::new_for_client_and_server_errors())
16        .start(tower::service_fn(noop_handle))
17        .unwrap();
18    let peer_id = network.connect(address).await?;
19    let peer = network.peer(peer_id).expect("just-connected peer is found");
20    Ok((network, peer))
21}

23pub fn random_key() -> [u8; 32] {
24    let mut rng = rand::thread_rng();
25    let mut bytes = [0u8; 32];
26    rand::RngCore::fill_bytes(&mut rng, &mut bytes[..]);
27    bytes
28}

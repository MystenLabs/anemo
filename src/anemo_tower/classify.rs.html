7/// Trait for classifying responses as either success or failure.
8///
9/// Response classifiers are used in cases where middleware needs to determine
10/// whether a response completed successfully or failed. For example, they may
11/// be used by logging or metrics middleware to record failures differently
12/// from successes.
13///
14/// Furthermore, when a response fails, a response classifier may provide
15/// additional information about the failure.
16pub trait Classifier {
17    /// The type returned when a response is classified as a failure.
18    ///
19    /// Depending on the classifier, this may simply indicate that the
20    /// request failed, or it may contain additional  information about
21    /// the failure, such as whether or not it is retryable.
22    type FailureClass;
23
24    /// Classify a response.
25    fn classify_response<B>(self, res: &Response<B>) -> Result<(), Self::FailureClass>;
26
27    /// Classify an error.
28    ///
29    /// Errors are always errors (doh) but sometimes it might be useful to have multiple classes of
30    /// errors. A retry policy might allow retrying some errors and not others.
31    fn classify_error<E>(self, error: &E) -> Self::FailureClass
32    where
33        E: fmt::Display + 'static;
34}
35
36/// Classifier that considers responses with a status code within some range to be failures.
37#[derive(Debug, Clone)]
38pub struct StatusInRangeAsFailures {
39    range: RangeInclusive<u16>,
40}
41
42impl StatusInRangeAsFailures {
43    /// Creates a new `StatusInRangeAsFailures` that classifies responses with a `5xx` status code
44    /// as failures, all others are considered successes.
45    pub fn new_for_server_errors() -> Self {
46        Self { range: 500..=599 }
47    }
48
49    /// Creates a new `StatusInRangeAsFailures` that classifies responses with a `4xx` or `5xx`
50    /// status code as failures, all others are considered successes.
51    pub fn new_for_client_and_server_errors() -> Self {
52        Self { range: 400..=599 }
53    }
54}
55
56impl Classifier for StatusInRangeAsFailures {
57    type FailureClass = StatusInRangeFailureClass;
58
59    fn classify_response<B>(self, response: &Response<B>) -> Result<(), Self::FailureClass> {
60        if self.range.contains(&response.status().to_u16()) {
61            let class = StatusInRangeFailureClass::StatusCode {
62                code: response.status(),
63                status_message: response
64                    .headers()
65                    .get(header::STATUS_MESSAGE)
66                    .map(ToOwned::to_owned),
67            };
68            Err(class)
69        } else {
70            Ok(())
71        }
72    }
73
74    fn classify_error<E>(self, error: &E) -> Self::FailureClass
75    where
76        E: fmt::Display + 'static,
77    {
78        StatusInRangeFailureClass::Error(error.to_string())
79    }
80}
81
82/// The failure class for [`StatusInRangeAsFailures`].
83#[derive(Debug)]
84pub enum StatusInRangeFailureClass {
85    /// A response was classified as a failure with the corresponding status.
86    StatusCode {
87        code: StatusCode,
88        status_message: Option<String>,
89    },
90    /// A response was classified as an error with the corresponding error description.
91    Error(String),
92}
93
94impl fmt::Display for StatusInRangeFailureClass {
95    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
96        match self {
97            Self::StatusCode {
98                code,
99                status_message,
100            } => {
101                write!(f, "Status code: {code}")?;
102                if let Some(status_message) = status_message {
103                    write!(f, " {status_message}")?;
104                }
105                Ok(())
106            }
107            Self::Error(error) => write!(f, "Error: {error}"),
108        }
109    }
110}

10/// The latency unit used to report latencies by middleware.
11#[non_exhaustive]
12#[derive(Copy, Clone, Debug)]
13pub enum LatencyUnit {
14    /// Use seconds.
15    Seconds,
16    /// Use milliseconds.
17    Millis,
18    /// Use microseconds.
19    Micros,
20    /// Use nanoseconds.
21    Nanos,
22}
23
24impl LatencyUnit {
25    pub fn display(self, latency: std::time::Duration) -> impl std::fmt::Display {
26        LatencyDisplay {
27            unit: self,
28            latency,
29        }
30    }
31}
32
33struct LatencyDisplay {
34    unit: LatencyUnit,
35    latency: std::time::Duration,
36}
37
38impl std::fmt::Display for LatencyDisplay {
39    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        match self.unit {
41            LatencyUnit::Seconds => write!(f, "{} s", self.latency.as_secs_f64()),
42            LatencyUnit::Millis => write!(f, "{} ms", self.latency.as_millis()),
43            LatencyUnit::Micros => write!(f, "{} μs", self.latency.as_micros()),
44            LatencyUnit::Nanos => write!(f, "{} ns", self.latency.as_nanos()),
45        }
46    }
47}

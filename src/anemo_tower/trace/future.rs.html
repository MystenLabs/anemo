14pin_project! {
15    /// Response future for [`Trace`].
16    ///
17    /// [`Trace`]: super::Trace
18    pub struct ResponseFuture<F, Classifier, OnResponse, OnFailure> {
19        #[pin]
20        pub(crate) inner: F,
21        pub(crate) classifier: Option<Classifier>,
22        pub(crate) span: Span,
23        pub(crate) on_response: Option<OnResponse>,
24        pub(crate) on_failure: Option<OnFailure>,
25        pub(crate) start: Instant,
26    }
27}
28
29impl<Fut, E, ClassifierT, OnResponseT, OnFailureT> Future
30    for ResponseFuture<Fut, ClassifierT, OnResponseT, OnFailureT>
31where
32    Fut: Future<Output = Result<Response<Bytes>, E>>,
33    E: std::fmt::Display + 'static,
34    ClassifierT: Classifier,
35    OnResponseT: OnResponse,
36    OnFailureT: OnFailure<ClassifierT::FailureClass>,
37{
38    type Output = Result<Response<Bytes>, E>;
39
40    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
41        let this = self.project();
42        let _guard = this.span.enter();
43        let result = futures::ready!(this.inner.poll(cx));
44        let latency = this.start.elapsed();
45        let classifier = this.classifier.take().unwrap();
46        let on_response = this.on_response.take().unwrap();
47        let mut on_failure = this.on_failure.take().unwrap();
48
49        match &result {
50            Ok(response) => {
51                on_response.on_response(response, latency, this.span);
52                if let Err(failure_class) = classifier.classify_response(response) {
53                    on_failure.on_failure(failure_class, latency, this.span);
54                }
55            }
56            Err(err) => {
57                let failure_class = classifier.classify_error(err);
58                on_failure.on_failure(failure_class, latency, this.span);
59            }
60        }
61
62        Poll::Ready(result)
63    }
64}

5/// [`Layer`] that adds high level [tracing] to a [`Service`].
6///
7/// See the [module docs](crate::trace) for more details.
8///
9/// [`Layer`]: tower::layer::Layer
10/// [tracing]: https://crates.io/crates/tracing
11/// [`Service`]: tower::Service
12#[derive(Debug, Copy, Clone)]
13pub struct TraceLayer<
14    Classifier,
15    MakeSpan = DefaultMakeSpan,
16    OnRequest = DefaultOnRequest,
17    OnResponse = DefaultOnResponse,
18    OnFailure = DefaultOnFailure,
19> {
20    pub(crate) classifier: Classifier,
21    pub(crate) make_span: MakeSpan,
22    pub(crate) on_request: OnRequest,
23    pub(crate) on_response: OnResponse,
24    pub(crate) on_failure: OnFailure,
25}
26
27impl<C> TraceLayer<C> {
28    /// Create a new [`TraceLayer`] using the given [`Classifier`].
29    pub fn new(classifier: C) -> Self
30    where
31        C: Classifier,
32    {
33        Self {
34            classifier,
35            make_span: DefaultMakeSpan::new(),
36            on_failure: DefaultOnFailure::default(),
37            on_request: DefaultOnRequest::default(),
38            on_response: DefaultOnResponse::default(),
39        }
40    }
41}

43impl TraceLayer<StatusInRangeAsFailures> {
44    /// Create a new [`TraceLayer`] using [`StatusInRangeAsFailures`] configured to classify server
45    /// errors as failures.
46    pub fn new_for_server_errors() -> Self {
47        Self::new(StatusInRangeAsFailures::new_for_server_errors())
48    }
49
50    /// Create a new [`TraceLayer`] using [`StatusInRangeAsFailures`] configured to classify client
51    /// and server errors as failures.
52    pub fn new_for_client_and_server_errors() -> Self {
53        Self::new(StatusInRangeAsFailures::new_for_client_and_server_errors())
54    }
55}

57impl<Classifier, MakeSpan, OnRequest, OnResponse, OnFailure>
58    TraceLayer<Classifier, MakeSpan, OnRequest, OnResponse, OnFailure>
59{
60    /// Customize what to do when a request is received.
61    ///
62    /// `NewOnRequest` is expected to implement [`OnRequest`].
63    ///
64    /// [`OnRequest`]: super::OnRequest
65    pub fn on_request<NewOnRequest>(
66        self,
67        new_on_request: NewOnRequest,
68    ) -> TraceLayer<Classifier, MakeSpan, NewOnRequest, OnResponse, OnFailure> {
69        TraceLayer {
70            classifier: self.classifier,
71            make_span: self.make_span,
72            on_request: new_on_request,
73            on_response: self.on_response,
74            on_failure: self.on_failure,
75        }
76    }
77
78    /// Customize what to do when a response has been produced.
79    ///
80    /// `NewOnResponse` is expected to implement [`OnResponse`].
81    ///
82    /// [`OnResponse`]: super::OnResponse
83    pub fn on_response<NewOnResponse>(
84        self,
85        new_on_response: NewOnResponse,
86    ) -> TraceLayer<Classifier, MakeSpan, OnRequest, NewOnResponse, OnFailure> {
87        TraceLayer {
88            classifier: self.classifier,
89            make_span: self.make_span,
90            on_request: self.on_request,
91            on_response: new_on_response,
92            on_failure: self.on_failure,
93        }
94    }
95
96    /// Customize what to do when a response has been classified as a failure.
97    ///
98    /// `NewOnFailure` is expected to implement [`OnFailure`].
99    ///
100    /// [`OnFailure`]: super::OnFailure
101    pub fn on_failure<NewOnFailure>(
102        self,
103        new_on_failure: NewOnFailure,
104    ) -> TraceLayer<Classifier, MakeSpan, OnRequest, OnResponse, NewOnFailure> {
105        TraceLayer {
106            classifier: self.classifier,
107            make_span: self.make_span,
108            on_request: self.on_request,
109            on_response: self.on_response,
110            on_failure: new_on_failure,
111        }
112    }
113
114    /// Customize how to make [`Span`]s that all request handling will be wrapped in.
115    ///
116    /// `NewMakeSpan` is expected to implement [`MakeSpan`].
117    ///
118    /// [`MakeSpan`]: super::MakeSpan
119    /// [`Span`]: tracing::Span
120    pub fn make_span_with<NewMakeSpan>(
121        self,
122        new_make_span: NewMakeSpan,
123    ) -> TraceLayer<Classifier, NewMakeSpan, OnRequest, OnResponse, OnFailure> {
124        TraceLayer {
125            classifier: self.classifier,
126            make_span: new_make_span,
127            on_request: self.on_request,
128            on_response: self.on_response,
129            on_failure: self.on_failure,
130        }
131    }
132}

134impl<S, Classifier, MakeSpan, OnRequest, OnResponse, OnFailure> Layer<S>
135    for TraceLayer<Classifier, MakeSpan, OnRequest, OnResponse, OnFailure>
136where
137    Classifier: Clone,
138    MakeSpan: Clone,
139    OnRequest: Clone,
140    OnResponse: Clone,
141    OnFailure: Clone,
142{
143    type Service = Trace<S, Classifier, MakeSpan, OnRequest, OnResponse, OnFailure>;
144
145    fn layer(&self, inner: S) -> Self::Service {
146        Trace {
147            inner,
148            classifier: self.classifier.clone(),
149            make_span: self.make_span.clone(),
150            on_request: self.on_request.clone(),
151            on_response: self.on_response.clone(),
152            on_failure: self.on_failure.clone(),
153        }
154    }

7/// Trait used to tell [`Trace`] what to do when a response has been produced.
8///
9/// See the [module docs](../trace/index.html#on_response) for details on exactly when the
10/// `on_response` callback is called.
11///
12/// [`Trace`]: super::Trace
13pub trait OnResponse {
14    /// Do the thing.
15    ///
16    /// `latency` is the duration since the request was received.
17    ///
18    /// `span` is the `tracing` [`Span`], corresponding to this request, produced by the closure
19    /// passed to [`TraceLayer::make_span_with`]. It can be used to [record field values][record]
20    /// that weren't known when the span was created.
21    ///
22    /// [`Span`]: https://docs.rs/tracing/latest/tracing/span/index.html
23    /// [record]: https://docs.rs/tracing/latest/tracing/span/struct.Span.html#method.record
24    /// [`TraceLayer::make_span_with`]: crate::trace::TraceLayer::make_span_with
25    fn on_response(self, response: &Response<Bytes>, latency: Duration, span: &Span);
26}
27
28impl OnResponse for () {
29    #[inline]
30    fn on_response(self, _: &Response<Bytes>, _: Duration, _: &Span) {}
31}
32
33impl<F> OnResponse for F
34where
35    F: FnOnce(&Response<Bytes>, Duration, &Span),
36{
37    fn on_response(self, response: &Response<Bytes>, latency: Duration, span: &Span) {
38        self(response, latency, span)
39    }
40}

42/// The default [`OnResponse`] implementation used by [`Trace`].
43///
44/// [`Trace`]: super::Trace
45#[derive(Clone, Debug)]
46pub struct DefaultOnResponse {
47    level: Level,
48    latency_unit: LatencyUnit,
49    include_headers: bool,
50}
51
52impl Default for DefaultOnResponse {
53    fn default() -> Self {
54        Self {
55            level: DEFAULT_MESSAGE_LEVEL,
56            latency_unit: LatencyUnit::Millis,
57            include_headers: false,
58        }
59    }
60}
61
62impl DefaultOnResponse {
63    /// Create a new `DefaultOnResponse`.
64    pub fn new() -> Self {
65        Self::default()
66    }
67
68    /// Set the [`Level`] used for [tracing events].
69    ///
70    /// Please note that while this will set the level for the tracing events
71    /// themselves, it might cause them to lack expected information, like
72    /// request method or path. You can address this using
73    /// [`DefaultMakeSpan::level`].
74    ///
75    /// Defaults to [`Level::DEBUG`].
76    ///
77    /// [tracing events]: https://docs.rs/tracing/latest/tracing/#events
78    /// [`DefaultMakeSpan::level`]: crate::trace::DefaultMakeSpan::level
79    pub fn level(mut self, level: Level) -> Self {
80        self.level = level;
81        self
82    }
83
84    /// Set the [`LatencyUnit`] latencies will be reported in.
85    ///
86    /// Defaults to [`LatencyUnit::Millis`].
87    pub fn latency_unit(mut self, latency_unit: LatencyUnit) -> Self {
88        self.latency_unit = latency_unit;
89        self
90    }
91
92    /// Include response headers on the [`Event`].
93    ///
94    /// By default headers are not included.
95    ///
96    /// [`Event`]: tracing::Event
97    pub fn include_headers(mut self, include_headers: bool) -> Self {
98        self.include_headers = include_headers;
99        self
100    }
101}

103impl OnResponse for DefaultOnResponse {
104    fn on_response(self, response: &Response<Bytes>, latency: Duration, _: &Span) {
105        let headers = self
106            .include_headers
107            .then(|| tracing::field::debug(response.headers()));
108
109        // This macro is needed, unfortunately, because `tracing::event!` requires the level
110        // argument to be static. Meaning we can't just pass `self.level`.
111        macro_rules! log_response {
112            ($level:expr) => {
113                tracing::event!(
114                    $level,
115                    latency = %self.latency_unit.display(latency),
116                    status = response.status().to_u16(),
117                    headers = headers,
118                    "finished processing request",
119                )
120            };
121        }
122
123        match self.level {
124            Level::ERROR => {
125                log_response!(Level::ERROR)
126            }
127            Level::WARN => {
128                log_response!(Level::WARN)
129            }
130            Level::INFO => {
131                log_response!(Level::INFO)
132            }
133            Level::DEBUG => {
134                log_response!(Level::DEBUG)
135            }
136            Level::TRACE => {
137                log_response!(Level::TRACE)
138            }
139        }
140    }
141}
